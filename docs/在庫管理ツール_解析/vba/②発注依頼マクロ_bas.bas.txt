'****************************************************************
'*　<発注状況検索>ボタン
'****************************************************************
Sub 発注状況検索ボタン()

    Call 画面停止有効
    Call シート表示("発注状況確認", "A3")
    Call オートフィルタ解除("発注状況確認")
    Call オートフィルタ設定("A2", "R2", "A3")
    Call データエリア削除("A3", "AZ500000", "A3")
    Call フラグセット("発注状況確認", "発注依頼", "T1")
    '*----------*
    Call DB_OPEN
    Call テーブルを開く("Q_発注状況確認データ", "依頼日 desc")
    If (エラー = "Y") Then
    Else
        Call 発注状況確認データ取り込み
    End If
    Call テーブルを閉じる
    Call DB_CLOSE
    '*----------*
    Call 画面停止無効
    
End Sub

'****************************************************************
'*　<削除>ボタン
'****************************************************************
Sub 削除ボタン_発注依頼()
    
    Call 画面停止有効
    Call シート保護解除
    Call 最終行チェック(2, 13, 2)
    If (存在チェック = "N") Then
        Call メッセージ(1)
    Else
        Call 対象データチェック(3, 最終行, "B", "削除")
        If (エラー = "Y") Then
            Call メッセージ(1)
        Else
            Call 発注依頼データバックアップ
            Call プルダウンセット("B3", "B1000", "削除", "A3")
        End If
    End If
    Call シート保護設定
    Call 画面停止無効
    
End Sub

'****************************************************************
'*　<依頼>ボタン
'****************************************************************
Sub 依頼ボタン()
    
    Call 画面停止有効
    Call シート保護解除
    Call 最終行チェック(3, 13, 2)
    If (存在チェック = "N") Then
        Call メッセージ(4)
    Else
        If MsgBox("発注依頼をしてもよろしいですか？", 292, "メッセージ") = 6 Then
            '*----------*
            Call DB_OPEN
            Call テーブルを開く("T_在庫発注データ", "コード")
            Call 発注依頼データ登録
            Call テーブルを閉じる
            Call DB_CLOSE
            '*----------*
            Call メッセージ(9)
            Call データエリア削除("A3", "R500000", "A3")
            Call プルダウンセット("B3", "B1000", "削除", "A3")
        End If
    End If
    Call シート保護設定
    Call 画面停止無効
    
End Sub

'================================================================
'=  発注依頼データ登録
'================================================================
Sub 発注依頼データ登録()
    
    With r
        For w行 = 3 To 最終行
            .AddNew
            ![コード] = 未入力の場合Null変換(Range("C" & w行))
            ![品名] = 未入力の場合Null変換(Range("D" & w行))
            ![数量] = 未入力の場合Null変換(Range("E" & w行))
            ![納期] = 未入力の場合Null変換(Range("F" & w行))
            ![型式] = 未入力の場合Null変換(Range("G" & w行))
            ![依頼者] = 未入力の場合Null変換(Range("H" & w行))
            ![依頼日] = 未入力の場合Null変換(Range("I" & w行))
            ![管理NO] = 未入力の場合Null変換(Range("J" & w行))
            ![備考] = 未入力の場合Null変換(Range("K" & w行))
            .Update
        Next
    End With

End Sub

'================================================================
'=  発注依頼データバックアップ
'================================================================
Sub 発注依頼データバックアップ()

    Dim 二次元配列 As Variant
    Dim 行 As Long
    Dim 列 As Long
    
    ReDim 二次元配列(最終行 - 1, 10)
    
    w行 = 0
    For 行 = 3 To 最終行
        If (Range("B" & 行) = "削除") Then
        Else
            w列 = 0
            For 列 = 1 To 11
                If (列 = 1) Then
                    二次元配列(w行, 0) = w行 + 1
                Else
                    二次元配列(w行, w列) = Cells(行, 列)
                End If
                w列 = w列 + 1
            Next
            w行 = w行 + 1
        End If
    Next
    Range("A3:AZ500000").Delete Shift:=xlUp
    If (w行 <= 1) Then
        Range("A3:K3") = 二次元配列
    Else
        Range("A3:K" & (w行 + 3)) = 二次元配列
    End If

End Sub

'================================================================
'=  発注登録画面を開く

Sub 発注登録画面を開く()
    
    Range("A3").Select
    発注登録画面.Show

End Sub
