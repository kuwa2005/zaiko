'****************************************************************
'*  画面がアクティブになった場合
'****************************************************************
Private Sub UserForm_Activate()
    
    Dim w日 As Long
        
    '*----------*
    コード = ""
    品名 = ""
    数量 = ""
    単位.Clear
    単価 = ""
    残数量 = ""
    安全在庫数 = ""
    最小発注数量 = ""
    標準納入日数 = ""
    取引先 = ""
    棚番 = ""
    備考 = ""
    '*----------*
    '（コード）
    コード = wコード
    '（品名）
    品名 = 未入力の場合Null変換(w品名)
    '（数量）
    数量 = 未入力の場合Null変換(w数量)
    '（単位）
    With 単位
        .AddItem "個"
        .AddItem "本"
        .AddItem "枚"
        .AddItem "件"
        .AddItem "m"
        .AddItem "kg"
    End With
    単位 = 未入力の場合Null変換(w単位)
    '（単価）
    単価 = 未入力の場合Null変換(w単価)
    '（残数量）
    残数量 = 未入力の場合Null変換(w残数量)
    '（安全在庫数）
    安全在庫数 = 未入力の場合Null変換(w安全在庫数)
    '（最小発注数量）
    最小発注数量 = 未入力の場合Null変換(w最小発注数量)
    '（標準納入日数）
    標準納入日数 = w標準納入日数
    '（取引先）
    取引先 = 未入力の場合Null変換(w取引先)
    '（棚版）
    棚番 = 未入力の場合Null変換(w棚番)
    '（備考）
    備考 = w備考
    '*----------*
    品名.SetFocus

End Sub

'****************************************************************
'*  <削除>ボタンがクリックされた場合
'****************************************************************
Private Sub 削除_Click()
    
    Call 画面停止有効
    'Call シート保護解除
    If MsgBox("削除してもよろしいですか？", 292, "メッセージ") = 6 Then
        '*----------*
        Call DB_OPEN
        Call データ削除("T_在庫マスタ", "コード", コード)
        Call DB_CLOSE
        '*----------*
        Call データエリア削除("A3", "AZ500000", "A3")
        '*----------*
        Call DB_OPEN
        Call テーブルを開く("T_在庫マスタ", "コード")
        If (エラー = "Y") Then
        Else
            Call 在庫マスタデータ取り込み
        End If
        Call テーブルを閉じる
        Call DB_CLOSE
        '*----------*
        Call 在庫マスタ変更画面を閉じる
    End If
    'Call シート保護設定
    Call 画面停止無効
    
End Sub

'****************************************************************
'*  <変更>ボタンがクリックされた場合
'****************************************************************
Private Sub 変更_Click()
    
    Call 画面停止有効
    'Call シート保護解除
    Call 在庫マスタ変更データチェック
    If (エラー = "Y") Then
    Else
        '*----------*
        Call DB_OPEN
        Call テーブルを開く("T_在庫マスタ", "コード")
        Call 在庫マスタ変更データセット
        Call テーブルを閉じる
        Call DB_CLOSE
        '*----------*
        Call データエリア削除("A3", "AZ500000", "A3")
        '*----------*
        Call DB_OPEN
        Call テーブルを開く("T_在庫マスタ", "コード")
        If (エラー = "Y") Then
        Else
            Call 在庫マスタデータ取り込み
        End If
        Call テーブルを閉じる
        Call DB_CLOSE
        '*----------*
        Call 在庫マスタ変更画面を閉じる
    End If
    'Call シート保護設定
    Call 画面停止無効
    
End Sub

'****************************************************************

'****************************************************************
Private Sub 閉じる_Click()
    
    Call 在庫マスタ変更画面を閉じる

End Sub

'================================================================
'=  在庫マスタ変更データセット
'================================================================
Sub 在庫マスタ変更データセット()
    
    With r
        .FindFirst "コード = '" & コード & "'"
        If (Not .NoMatch) Then
            .Edit
            ![コード] = 未入力の場合Null変換(コード)
            ![品名] = 未入力の場合Null変換(品名)
            ![数量] = 数量
            ![単位] = 未入力の場合Null変換(単位)
            ![単価] = 未入力の場合Null変換(単価)
            ![残数量] = 残数量
            ![安全在庫数] = スペースの場合ゼロ変換(安全在庫数)
            ![最小発注数量] = スペースの場合ゼロ変換(最小発注数量)
            ![棚番] = 未入力の場合Null変換(棚番)
            ![標準納入日数] = 未入力の場合Null変換(標準納入日数)
            ![取引先] = 未入力の場合Null変換(取引先)
            ![備考] = 未入力の場合Null変換(備考)
            ![登録者] = Sheets("メイン").Range("N5")
            ![登録日] = Format(Now, "yyyy/mm/dd")
            ![備考] = 未入力の場合Null変換(備考)
            .Update
        End If
    End With

End Sub

'================================================================
'=　在庫マスタ変更データチェック
'================================================================
Sub 在庫マスタ変更データチェック()
    
    エラー = "N"
    
    '*----------*
    If (Trim(品名) = "") Then
        MsgBox "品名を入力して下さい。", 16, "メッセージ"
        品名.SetFocus
        エラー = "Y"
        Exit Sub
    End If
    '*----------*
    If (Trim(数量) = "" Or 数量 = 0) Then
        MsgBox "基本数量を入力して下さい。", 16, "メッセージ"
        数量.SetFocus
        エラー = "Y"
        Exit Sub
    End If
    '*----------*
    If (Trim(単位) = "") Then
        MsgBox "単位を入力して下さい。", 16, "メッセージ"
        単位.SetFocus
        エラー = "Y"
        Exit Sub
    End If
    '*----------*
    If (Trim(単価) = "" Or 単価 = 0) Then
        MsgBox "単価を入力して下さい。", 16, "メッセージ"
        単価.SetFocus
        エラー = "Y"
        Exit Sub
    End If
    '*----------*
    If (Trim(残数量) = "") Then
        MsgBox "残数量を入力して下さい。", 16, "メッセージ"
        残数量.SetFocus
        エラー = "Y"
        Exit Sub
    End If
    '*----------*
    棚番 = 半角変換(大文字変換(Trim(棚番)))
    '*----------*
    
End Sub

'================================================================
'=　在庫マスタ変更画面を閉じる
'================================================================
Sub 在庫マスタ変更画面を閉じる()
    
    Range("A3").Select
    Unload 在庫マスタ変更画面
    
End Sub
