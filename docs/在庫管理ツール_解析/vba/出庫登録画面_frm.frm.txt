Attribute VB_GlobalNameSpace = False
Attribute VB_Creatable = False
Attribute VB_PredeclaredId = True
Attribute VB_Exposed = False
Attribute VB_TemplateDerived = False
Attribute VB_Customizable = False
Option Explicit

'****************************************************************
'*  画面がアクティブになった場合
'****************************************************************
Private Sub UserForm_Activate()
    
    '*----------*
    コード = ""
    品名 = ""
    出庫数量 = ""
    備考 = ""
    '*----------*
    コード = wコード
    品名 = w品名
    '*----------*
    出庫数量.SetFocus

End Sub

'****************************************************************
'*  <登録>ボタンがクリックされた場合
'****************************************************************
Private Sub 登録_Click()
    
    Call 画面停止有効
    Call 出庫データチェック
    If (エラー = "Y") Then
    Else
        If MsgBox("登録してもよろしいですか？", 292, "メッセージ") = 6 Then
            '*----------*
            Call DB_OPEN
            Call テーブルを開く("T_採番テーブル", "出庫管理NO")
            Call 出庫管理NO採番
            Call テーブルを閉じる
            Call DB_CLOSE
            '*----------*
            Call DB_OPEN
            Call テーブルを開く("T_出庫データ", "コード")
            Call 出庫データセット
            Call テーブルを閉じる
            Call DB_CLOSE
            '*----------*
            Call メッセージ(6)
            Call 出庫登録画面を閉じる
        End If
    End If
    Call 画面停止無効
    
End Sub

'****************************************************************
'*  <閉じる>ボタンがクリックされた場合
'****************************************************************
Private Sub 閉じる_Click()
    
    Call 出庫登録画面を閉じる

End Sub

'================================================================
'=　出庫データチェック
'================================================================
Sub 出庫データチェック()
    
    エラー = "N"
    
    If (IsNull(出庫数量) = True Or Trim(出庫数量) = "" Or 出庫数量 = 0) Then
        MsgBox "出庫数量が入力されていません。", 16, "メッセージ"
        出庫数量.SetFocus
        エラー = "Y"
        Exit Sub
    End If
    
End Sub

'================================================================
'=　出庫登録画面を閉じる
'================================================================
Sub 出庫登録画面を閉じる()
    
    Range("A3").Select
    Unload 出庫登録画面
    
End Sub

'================================================================
'=  出庫管理NO採番
'================================================================
Sub 出庫管理NO採番()
    
    With r
        .MoveFirst
        w管理NO = ![出庫管理NO] + 1
        .Edit
        ![出庫管理NO] = w管理NO
        .Update
    End With

End Sub
    
'================================================================
'=  出庫データセット
'================================================================
Sub 出庫データセット()
    
    With r
        .AddNew
        ![コード] = 未入力の場合Null変換(コード)
        ![品名] = 未入力の場合Null変換(品名)
        ![出庫数] = 未入力の場合Null変換(出庫数量)
        ![備考] = 未入力の場合Null変換(備考)
        ![出庫者] = Sheets("メイン").Range("N5")
        ![出庫日] = Format(Now, "yyyy/mm/dd")
        ![管理NO] = CStr("S" & Format(w管理NO, "0000000000"))
        .Update
    End With

End Sub
