'================================================================
'=  在庫発注データ情報退避
'================================================================
Sub 在庫発注データ情報退避()

    If (r.EOF) Then
        Exit Sub
    End If
    
    With r
        w注番 = ![注番]
        wコード = ![コード]
        w品名 = ![品名]
        w数量 = ![数量]
        w納期 = ![納期]
        w型式 = ![型式]
        w依頼者 = ![依頼者]
        w依頼日 = ![依頼日]
        w受付者 = ![受付者]
        w受付日 = ![受付日]
        w備考 = ![備考]
    End With
            
End Sub

'================================================================
'=  手配データ分割画面を開く
'================================================================
Sub 手配データ分割画面を開く()

    Range("A3").Select
    手配データ分割画面.Show

End Sub

'================================================================
'=  管理NO退避
'================================================================
Sub 管理NO退避()

    エラー = "N"

    If (Trim(Range("O" & w行)) = "") Then
        エラー = "Y"
    Else
        w管理NO = 大文字変換(半角変換(Trim(Range("O" & w行))))
    End If
    
End Sub
