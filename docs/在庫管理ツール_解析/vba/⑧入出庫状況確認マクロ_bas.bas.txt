'****************************************************************
'*　<一括変更>ボタン
'****************************************************************
Sub 一括変更ボタン()

    If MsgBox("前月残数の一括更新を行います。宜しいですか？" & Chr(13) & Chr(13) & "※実行すると現在登録されている前月残数が一括で更新されます。", 308, "メッセージ") = 6 Then
        Call 画面停止有効
        'Call シート保護解除
        Call 最終行チェック(1, 1, 3)
        If (存在チェック = "N") Then
        Else
            '*----------*
            Call DB_OPEN
            Call テーブルを開く("T_在庫マスタ", "コード")
            Call 前月残数一括更新
            Call テーブルを閉じる
            Call DB_CLOSE
            '*----------*
            Call メッセージ(8)
        End If
        'Call シート保護設定
        Call 画面停止無効
    End If

End Sub

'================================================================
'=　前月残数一括更新
'================================================================
Sub 前月残数一括更新()
    
    With r
        For w行 = 4 To 最終行
            .FindFirst "コード = '" & Trim(Range("B" & w行)) & "'"
            If (Not .NoMatch) Then
                .Edit
                ![残数量] = Trim(Range("I" & w行))
                .Update
            End If
        Next
    End With

End Sub

'================================================================
'=　入出庫状況シートヘッダーセット
'================================================================
Sub 入出庫状況シートヘッダーセット()
    
    Range("A1") = Year(Now()) & "/" & Month(Now) & " 入出庫状況確認"

End Sub

'================================================================
'=　入出庫確認データ取り込み
'================================================================
Sub 入出庫確認データ取り込み()

    Dim 二次元配列 As Variant
    Dim w基本数量 As Double
    Dim w前残数量 As Double
    Dim w前残金額 As Double
    Dim w当入数量 As Double
    Dim w当入金額 As Double
    Dim w当出数量 As Double
    Dim w当出金額 As Double
    Dim w当残数量 As Double
    Dim w当残金額 As Double
    Dim w前残数量_合計 As Double
    Dim w前残金額_合計 As Double
    Dim w当入数量_合計 As Double
    Dim w当入金額_合計 As Double
    Dim w当出数量_合計 As Double
    Dim w当出金額_合計 As Double
    Dim w当残数量_合計 As Double
    Dim w当残金額_合計 As Double
        
    If (r.EOF) Then
        Exit Sub
    End If
    
    w前残数量_合計 = 0
    w前残金額_合計 = 0
    w当入数量_合計 = 0
    w当入金額_合計 = 0
    w当出数量_合計 = 0
    w当出金額_合計 = 0
    w当残数量_合計 = 0
    w当残金額_合計 = 0
    
    ReDim 二次元配列(全件数 - 1, 16)
    
    w行 = 0
    With r
        .MoveFirst
        Do Until (.EOF)
        
            '*----------*
            w基本数量 = Nullの場合ゼロ変換(![数量])
            w前残数量 = Nullの場合ゼロ変換(![残数量])
            w前残金額 = (w前残数量 * Nullの場合ゼロ変換(![単価])) / w基本数量
            w当入数量 = Nullの場合ゼロ変換(![当月入庫数])
            w当入金額 = (w当入数量 * Nullの場合ゼロ変換(![単価])) / w基本数量
            w当出数量 = Nullの場合ゼロ変換(![当月出庫数])
            w当出金額 = (w当出数量 * Nullの場合ゼロ変換(![単価])) / w基本数量
            w当残数量 = w前残数量 + w当入数量 - w当出数量
            w当残金額 = w前残金額 + w当入金額 - w当出金額
            '*----------*
            w前残数量_合計 = w前残数量_合計 + w前残数量
            w前残金額_合計 = w前残金額_合計 + w前残金額
            w当入数量_合計 = w当入数量_合計 + w当入数量
            w当入金額_合計 = w当入金額_合計 + w当入金額
            w当出数量_合計 = w当出数量_合計 + w当出数量
            w当出金額_合計 = w当出金額_合計 + w当出金額
            w当残数量_合計 = w当残数量_合計 + w当残数量
            w当残金額_合計 = w当残金額_合計 + w当残金額
            '*----------*
            
            二次元配列(w行, 0) = w行 + 1
            二次元配列(w行, 1) = ![コード]
            二次元配列(w行, 2) = ![品名]
            二次元配列(w行, 3) = ![数量]
            二次元配列(w行, 4) = ![単位]
            二次元配列(w行, 5) = ![単価]
            二次元配列(w行, 6) = ![安全在庫数]
            二次元配列(w行, 7) = ![棚番]
            二次元配列(w行, 8) = w前残数量
            二次元配列(w行, 9) = w前残金額
            二次元配列(w行, 10) = w当入数量
            二次元配列(w行, 11) = w当入金額
            二次元配列(w行, 12) = w当出数量
            二次元配列(w行, 13) = w当出金額
            二次元配列(w行, 14) = w当残数量
            二次元配列(w行, 15) = w当残金額
            If (w当残数量 = 0) Then
                二次元配列(w行, 16) = "在庫が0です。確認して下さい。"
            Else
                If (w当残数量 < ![安全在庫数]) Then
                    二次元配列(w行, 16) = "安全在庫以下です。発注して下さい。"
                End If
            End If
            w行 = w行 + 1
            .MoveNext
        Loop
    End With
    If (w行 <= 1) Then
        Range("A4:Q4") = 二次元配列
    Else
        Range("A4:Q" & w行 + 3) = 二次元配列
    End If
    
    '*----------*
    Range("H1") = "合計"
    Range("I1") = w前残数量_合計
    Range("J1") = w前残金額_合計
    Range("K1") = w当入数量_合計
    Range("L1") = w当入金額_合計
    Range("M1") = w当出数量_合計
    Range("N1") = w当出金額_合計
    Range("O1") = w当残数量_合計
    Range("P1") = w当残金額_合計
    '*----------*

End Sub

'================================================================
'=　発注予定情報セット
'================================================================
Sub 発注予定情報セット()
    
    With r
        For w行 = 4 To 最終行
            .FindFirst "コード = '" & Trim(Range("B" & w行)) & "'"
            If (Not .NoMatch) Then
                Range("Q" & w行) = "発注予定又は発注済みです。"
            End If
        Next
    End With

End Sub
