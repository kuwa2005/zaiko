Sub 出庫確認ボタン()
    
    Call 画面停止有効
    Call シート表示("出庫確認", "A3")
    'Call シート保護解除
    Call オートフィルタ解除("出庫確認")
    Call オートフィルタ設定("A2", "H2", "A3")
    Call データエリア削除("A3", "AZ500000", "A3")
    '*----------*
    Call DB_OPEN
    Call テーブルを開く("T_出庫データ", "管理NO desc")
    If (エラー = "Y") Then
    Else
        Call 出庫実績確認データ取り込み
    End If
    Call テーブルを閉じる
    Call DB_CLOSE
    '*----------*
    'Call シート保護設定
    Call 画面停止無効

End Sub

'================================================================
'=  出庫データ取り込み
'================================================================
Sub 出庫データ取り込み()

    Dim 二次元配列 As Variant
    
    エラー = "N"
    
    If (r.EOF) Then
        エラー = "Y"
        Exit Sub
    End If
    
    ReDim 二次元配列(全件数 - 1, 8)
        
    w行 = 0
    With r
        .MoveFirst
        Do Until (.EOF)
            二次元配列(w行, 0) = w行 + 1
            二次元配列(w行, 1) = ![コード]
            二次元配列(w行, 2) = ![品名]
            二次元配列(w行, 3) = ![数量]
            二次元配列(w行, 4) = ![単位]
            二次元配列(w行, 5) = ![安全在庫数]
            二次元配列(w行, 6) = ![標準納入日数]
            二次元配列(w行, 7) = ![棚番]
            二次元配列(w行, 8) = ![備考]
            w行 = w行 + 1
            .MoveNext
        Loop
    End With
    If (w行 <= 1) Then
        Range("A3:I3") = 二次元配列
    Else
        Range("A3:I" & w行 + 2) = 二次元配列
    End If

End Sub
