'****************************************************************
'*　<発注依頼>ボタン
'****************************************************************
Sub 発注依頼ボタン()
    
    Call 画面停止有効
    Call 担当者チェック
    If (エラー = "Y") Then
    Else
        Call シート表示("在庫マスタ", "A3")
        'Call シート保護解除
        Call オートフィルタ解除("在庫マスタ")
        Call オートフィルタ設定("A2", "O2", "A3")
        Call データエリア削除("A3", "AZ500000", "A3")
        '*----------*
        Call DB_OPEN
        Call テーブルを開く("T_在庫マスタ", "コード")
        If (エラー = "Y") Then
        Else
            Call 在庫マスタデータ取り込み
        End If
        Call テーブルを閉じる
        Call DB_CLOSE
        '*----------*
        'Call シート保護設定
        Call シート表示("発注依頼", "A3")
        Call シート保護解除
        Call オートフィルタ解除("発注依頼")
        Call オートフィルタ設定("A2", "K2", "A3")
        Call データエリア削除("A3", "AZ500000", "A3")
        Call プルダウンセット("B3", "B1000", "削除", "A3")
        Call シート保護設定
    End If
    Call 画面停止無効
    
End Sub
