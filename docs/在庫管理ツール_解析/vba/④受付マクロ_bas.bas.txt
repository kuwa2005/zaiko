'****************************************************************
'*　<印刷>ボタン
'****************************************************************
Sub 印刷ボタン()
    
    Call 画面停止有効
    'Call シート保護解除
    Call オートフィルタ解除("受付")
    Call オートフィルタ設定("A2", "O2", "A3")
    Call 最終行チェック(1, 15, 2)
    If (存在チェック = "N") Then
        Call メッセージ(5)
    Else
        Call 対象データチェック(3, 最終行, "B", "出力")
        If (エラー = "Y") Then
            Call メッセージ(5)
        Else
            If MsgBox("印刷を実行してもよろしいですか？", 292, "メッセージ") = 6 Then
                Call シート表示("受付", "A3")
                Call 最終行チェック(1, 15, 2)
                '*----------*
                Call DB_OPEN
                Call テーブルを開く("T_在庫発注データ", "依頼日 desc")
                Call 受付者及び受付日セット
                Call テーブルを閉じる
                Call DB_CLOSE
                '*----------*
                Call シート表示("発注依頼表", "A5")
                Call データエリア削除("A5", "AZ500000", "A5")
                Call シート表示("受付", "A3")
                Call 印刷データセット
                Call シート表示("発注依頼表", "A5")
                Call 最終行チェック(1, 9, 4)
                Call 印刷("発注依頼表")
                Call シート表示("受付", "A3")
                Call データエリア削除("A3", "AZ500000", "A3")
                '*----------*
                Call DB_OPEN
                Call テーブルを開く("T_在庫発注データ where isnull(受付日) = true ", "依頼日 desc")
                If (エラー = "Y") Then
                Else
                    Call 受付データ取り込み
                    Call プルダウンセット("B3", "B1000", "出力,削除", "A3")
                End If
                Call テーブルを閉じる
                Call DB_CLOSE
                '*----------*
                Call メッセージ(10)
            End If
        End If
    End If
    'Call シート保護設定
    Call 画面停止無効
    
End Sub

'================================================================
'=　受付変更データ保存
'================================================================
Sub 受付変更データ保存()
    
    With r
        For w行 = 3 To 最終行
            .FindFirst "管理NO = '" & Trim(Range("N" & w行)) & "'"
            If (Not .NoMatch) Then
                .Edit
                ![数量] = Range("E" & w行)
                ![納期] = Range("F" & w行)
                ![型式] = Range("G" & w行)
                ![備考] = Range("O" & w行)
                .Update
            End If
        Next
    End With

End Sub

'================================================================
'=　受付データ削除
'================================================================
Sub 受付データ削除()
    
    For w行 = 3 To 最終行
        If (Trim(Range("B" & w行)) = "削除") Then
            sql = "delete * from T_在庫発注データ where 管理NO  = '" & Trim(Range("N" & w行)) & "'"
            cn.Execute (sql)
        End If
    Next

End Sub

'================================================================
'=　印刷データセット
'================================================================
Sub 印刷データセット()
    
    Dim 行 As Long
    Dim 二次元配列 As Variant
    
    ReDim 二次元配列(最終行 - 1, 7)
    
    w行 = 0
    For 行 = 3 To 最終行
        If (Trim(Range("B" & 行)) = "出力") Then
            二次元配列(w行, 0) = w行 + 1          'NO
            二次元配列(w行, 1) = Range("C" & 行)  'コード
            二次元配列(w行, 2) = Range("D" & 行)  '品名
            二次元配列(w行, 3) = Range("E" & 行)  '数量
            二次元配列(w行, 4) = Range("F" & 行)  '納期
            二次元配列(w行, 5) = Range("G" & 行)  '型式
            二次元配列(w行, 6) = Range("N" & 行)  '管理NO
            w行 = w行 + 1
        End If
    Next
    If (w行 <= 1) Then
        Sheets("発注依頼表").Range("A5:G5") = 二次元配列
    Else
        Sheets("発注依頼表").Range("A5:G" & w行 + 4) = 二次元配列
    End If
    
End Sub

'================================================================
'=  受付者及び受付日セット
'================================================================
Sub 受付者及び受付日セット()
    
    With r
        For w行 = 3 To 最終行
            If (Trim(Range("B" & w行)) = "出力") Then
                .FindFirst "管理NO = '" & Trim(Range("N" & w行)) & "'"
                If (Not .NoMatch) Then
                    .Edit
                    ![受付者] = Sheets("メイン").Range("N5")
                    ![受付日] = Format(Now, "yyyy/mm/dd")
                    .Update
                End If
            End If
        Next
    End With

End Sub

'================================================================
'=  受付データ取り込み
'================================================================
Sub 受付データ取り込み()

    Dim 二次元配列 As Variant
    
    エラー = "N"
    
    If (r.EOF) Then
        エラー = "Y"
        Exit Sub
    End If
    
    ReDim 二次元配列(全件数 - 1, 14)
        
    w行 = 0
    With r
        .MoveFirst
        Do Until (.EOF)
            二次元配列(w行, 0) = w行 + 1
            二次元配列(w行, 2) = ![コード]
            二次元配列(w行, 3) = ![品名]
            二次元配列(w行, 4) = ![数量]
            二次元配列(w行, 5) = ![納期]
            二次元配列(w行, 6) = ![型式]
            二次元配列(w行, 7) = ![依頼者]
            二次元配列(w行, 8) = ![依頼日]
            二次元配列(w行, 9) = ![受付者]
            二次元配列(w行, 10) = ![受付日]
            二次元配列(w行, 11) = ![入庫者]
            二次元配列(w行, 12) = ![入庫日]
            二次元配列(w行, 13) = ![管理NO]
            二次元配列(w行, 14) = ![備考]
            w行 = w行 + 1
            .MoveNext
        Loop
    End With
    If (w行 <= 1) Then
        Range("A3:O3") = 二次元配列
    Else
        Range("A3:O" & w行 + 2) = 二次元配列
    End If

End Sub
