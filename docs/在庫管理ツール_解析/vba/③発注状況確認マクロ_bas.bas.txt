'****************************************************************
'*　<コード>ボタン
'****************************************************************
Sub コードボタン_発注状況確認()
    
    Call 画面停止有効
    Call オートフィルタ解除("発注状況確認")
    Call オートフィルタ設定("A2", "R2", "A3")
    Call 最終行チェック(1, 18, 2)
    If (存在チェック = "N") Then
    Else
        Call 並べ替え("B3", "AZ" & 最終行, "B3", "無", "無", "A3")
    End If
    Call 画面停止無効
    
End Sub

'****************************************************************
'*　<管理NO>ボタン
'****************************************************************
Sub 管理NOボタン_発注状況確認()
    
    Call 画面停止有効
    Call オートフィルタ解除("発注状況確認")
    Call オートフィルタ設定("A2", "R2", "A3")
    Call 最終行チェック(1, 18, 2)
    If (存在チェック = "N") Then
    Else
        Call 降順並べ替え("B3", "AZ" & 最終行, "Q3", "無", "無", "A3")
    End If
    Call 画面停止無効
    
End Sub

'****************************************************************
'*　<戻る>ボタン
'****************************************************************
Sub 発注状況確認戻るボタン()
    
    Call 戻りシート判定
    
End Sub

'================================================================
'=  戻りシート判定
'================================================================
Sub 戻りシート判定()
    
    Select Case Range("T1")
        Case "メイン"
            Sheets("メイン").Select
            Range("A1").Select
        Case "発注依頼"
            Sheets("発注依頼").Select
            Range("A3").Select
    End Select

End Sub

'================================================================
'=  発注状況確認データ取り込み
'================================================================
Sub 発注状況確認データ取り込み()

    Dim 二次元配列 As Variant
    
    エラー = "N"
    
    If (r.EOF) Then
        エラー = "Y"
        Exit Sub
    End If
    
    ReDim 二次元配列(全件数 - 1, 17)
        
    w行 = 0
    With r
        .MoveFirst
        Do Until (.EOF)
            二次元配列(w行, 0) = w行 + 1
            二次元配列(w行, 1) = ![コード]
            二次元配列(w行, 2) = ![品名]
            二次元配列(w行, 3) = ![数量]
            二次元配列(w行, 4) = ![基本数量]
            二次元配列(w行, 5) = ![単位]
            二次元配列(w行, 6) = ![単価]
            二次元配列(w行, 7) = (![数量] / ![基本数量]) * ![単価]
            二次元配列(w行, 8) = ![納期]
            二次元配列(w行, 9) = ![型式]
            二次元配列(w行, 10) = ![依頼者]
            二次元配列(w行, 11) = ![依頼日]
            二次元配列(w行, 12) = ![受付者]
            二次元配列(w行, 13) = ![受付日]
            二次元配列(w行, 14) = ![入庫者]
            二次元配列(w行, 15) = ![入庫日]
            二次元配列(w行, 16) = ![管理NO]
            二次元配列(w行, 17) = ![備考]
            w行 = w行 + 1
            .MoveNext
        Loop
    End With
    If (w行 <= 1) Then
        Range("A3:R3") = 二次元配列
    Else
        Range("A3:R" & w行 + 2) = 二次元配列
    End If

End Sub
