Sub 新規ボタン()
    
    Call 画面停止有効
    'Call シート保護解除
    Call オートフィルタ解除("在庫マスタ")
    Call オートフィルタ設定("A2", "O2", "A3")
    Call 在庫マスタ登録画面を開く
    'Call シート保護設定
    Call 画面停止無効

End Sub

'================================================================
'=　在庫マスタ登録画面を開く
'================================================================
Sub 在庫マスタ登録画面を開く()
    
    Range("A3").Select
    在庫マスタ登録画面.Show

End Sub

'================================================================
'=  在庫マスタデータ取り込み
'================================================================
Sub 在庫マスタデータ取り込み()

    Dim 二次元配列 As Variant
    
    エラー = "N"
    
    If (r.EOF) Then
        エラー = "Y"
        Exit Sub
    End If
    
    ReDim 二次元配列(全件数 - 1, 14)
        
    w行 = 0
    With r
        .MoveFirst
        Do Until (.EOF)
            二次元配列(w行, 0) = w行 + 1
            二次元配列(w行, 1) = ![コード]
            二次元配列(w行, 2) = ![品名]
            二次元配列(w行, 3) = ![数量]
            二次元配列(w行, 4) = ![単位]
            二次元配列(w行, 5) = ![単価]
            二次元配列(w行, 6) = ![残数量]
            二次元配列(w行, 7) = ![安全在庫数]
            二次元配列(w行, 8) = ![最小発注数量]
            二次元配列(w行, 9) = ![標準納入日数]
            二次元配列(w行, 10) = ![棚番]
            二次元配列(w行, 11) = ![取引先]
            二次元配列(w行, 12) = ![登録者]
            二次元配列(w行, 13) = ![登録日]
            二次元配列(w行, 14) = ![備考]
            w行 = w行 + 1
            .MoveNext
        Loop
    End With
    If (w行 <= 1) Then
        Range("A3:O3") = 二次元配列
    Else
        Range("A3:O" & w行 + 2) = 二次元配列
    End If

End Sub
