    Call 画面停止有効
    Call シート表示("メイン", "A1")
    Call シートクリア("メイン", "N5", "N5", "A1")
    Call 画面停止無効

End Sub

'================================================================
'=  対象データチェック
'================================================================
Sub 対象データチェック(開始行, 終了行, 列, 文字)

    エラー = "Y"
    
    For w行 = 開始行 To 終了行
        If (Trim(Range(列 & w行)) = 文字) Then
            エラー = "N"
            Exit Sub
        End If
    Next
    
End Sub

'================================================================
'=　プルダウンセット
'================================================================

    Range(開始セル & ":" & 終了セル).Validation.Add Type:=xlValidateList, AlertStyle:=xlValidAlertStop, Operator:=xlBetween, Formula1:=文字

End Sub

    エラー = "N"
    
    Set r = cn.OpenRecordset("select * from " & テーブル名 & " order by " & 項目名, dbOpenDynaset)
    If (r.EOF) Then
        エラー = "Y"
    Else
        r.MoveLast
        全件数 = r.RecordCount
    End If

End Sub

'================================================================
'=  並べ替え
'================================================================
Sub 並べ替え(開始範囲, 終了範囲, KEY①, KEY②, KEY③, セル)
    
    Range(開始範囲 & ":" & 終了範囲).Select
    If (KEY② = "無") Then
        If (KEY③ = "無") Then
            Selection.Sort Key1:=Range(KEY①), Order1:=xlAscending
        Else
            Selection.Sort Key1:=Range(KEY①), Order1:=xlAscending, _
                           Key2:=Range(KEY③), Order2:=xlAscending
        End If
    Else
        If (KEY③ = "無") Then
            Selection.Sort Key1:=Range(KEY①), Order1:=xlAscending, _
                           Key2:=Range(KEY②), Order2:=xlAscending
        Else
            Selection.Sort Key1:=Range(KEY①), Order1:=xlAscending, _
                           Key2:=Range(KEY②), Order2:=xlAscending, _
                           Key3:=Range(KEY③), Order3:=xlAscending
        End If
    End If
    Application.Goto Range(セル), True

End Sub

'================================================================
'=  降順並べ替え
'================================================================
Sub 降順並べ替え(開始範囲, 終了範囲, KEY①, KEY②, KEY③, セル)
    
    Range(開始範囲 & ":" & 終了範囲).Select
    If (KEY② = "無") Then
        If (KEY③ = "無") Then
            Selection.Sort Key1:=Range(KEY①), Order1:=xlDescending
        Else
            Selection.Sort Key1:=Range(KEY①), Order1:=xlDescending, _
                           Key2:=Range(KEY③), Order2:=xlAscending
        End If
    Else
        If (KEY③ = "無") Then
            Selection.Sort Key1:=Range(KEY①), Order1:=xlDescending, _
                           Key2:=Range(KEY②), Order2:=xlAscending
        Else
            Selection.Sort Key1:=Range(KEY①), Order1:=xlDescending, _
                           Key2:=Range(KEY②), Order2:=xlAscending, _
                           Key3:=Range(KEY③), Order3:=xlAscending
        End If
    End If
    Range(セル).Select

End Sub

'================================================================
'=  最終行チェック
'================================================================
Sub 最終行チェック(開始列, 終了列, 項目行)

    Dim w最終行 As Long
    
    存在チェック = "Y"
    
    最終行 = ActiveSheet.Cells(500000, 開始列).End(xlUp).Row
    For w列 = 開始列 To 終了列
        w最終行 = ActiveSheet.Cells(500000, w列).End(xlUp).Row
        If (最終行 < w最終行) Then
            最終行 = w最終行
        End If
    Next
    If (項目行 = 最終行) Then
        存在チェック = "N"
    End If

End Sub
