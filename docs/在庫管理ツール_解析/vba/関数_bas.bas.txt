'================================================================
'=  未入力の場合Null変換
'================================================================
Function 未入力の場合Null変換(変数)
    
    If (変数 = "") Then
        未入力の場合Null変換 = Null
    Else
        未入力の場合Null変換 = 変数
    End If

End Function

'================================================================
'=  Nullの場合ゼロ変換
'================================================================
Function Nullの場合ゼロ変換(変数)
    
    If (IsNull(変数) = True) Then
        Nullの場合ゼロ変換 = 0
    Else
        Nullの場合ゼロ変換 = 変数
    End If

End Function

'================================================================
'=  スペースの場合ゼロ変換
'================================================================
Function スペースの場合ゼロ変換(変数)
    
    If (Trim(変数) = "") Then
        スペースの場合ゼロ変換 = 0
    Else
        スペースの場合ゼロ変換 = 変数
    End If

End Function

'================================================================
'=  大文字変換
'================================================================
Function 大文字変換(文字)
    
    大文字変換 = StrConv(文字, vbUpperCase)

End Function

'================================================================
'=  半角変換
'================================================================
Function 半角変換(文字)
    
    半角変換 = StrConv(文字, vbNarrow)

End Function
