'****************************************************************
'*　<コード>ボタン
'****************************************************************
Sub コードボタン_出庫確認()
    
    Call 画面停止有効
    Call オートフィルタ解除("出庫確認")
    Call オートフィルタ設定("A2", "H2", "A3")
    Call 最終行チェック(1, 9, 2)
    Call 並べ替え("B3", "AZ" & 最終行, "B3", "無", "無", "A3")
    Call 画面停止無効
    
End Sub

'****************************************************************
'*　<出庫日>ボタン
'****************************************************************
Sub 出庫日ボタン()
    
    Call 画面停止有効
    Call オートフィルタ解除("出庫確認")
    Call オートフィルタ設定("A2", "H2", "A3")
    Call 最終行チェック(1, 9, 2)
    Call 降順並べ替え("B3", "AZ" & 最終行, "F3", "B3", "無", "A3")
    Call 画面停止無効
    
End Sub

'================================================================
'=  出庫実績確認データ取り込み
'================================================================
Sub 出庫実績確認データ取り込み()

    Dim 二次元配列 As Variant
    
    If (r.EOF) Then
        Exit Sub
    End If
    
    ReDim 二次元配列(全件数 - 1, 7)
        
    w行 = 0
    With r
        .MoveFirst
        Do Until (.EOF)
            二次元配列(w行, 0) = w行 + 1
            二次元配列(w行, 1) = ![コード]
            二次元配列(w行, 2) = ![品名]
            二次元配列(w行, 3) = ![出庫数]
            二次元配列(w行, 4) = ![出庫者]
            二次元配列(w行, 5) = ![出庫日]
            二次元配列(w行, 6) = ![管理NO]
            二次元配列(w行, 7) = ![備考]
            w行 = w行 + 1
            .MoveNext
        Loop
    End With
    If (w行 = 1) Then
        Range("A3:H3") = 二次元配列
    Else
        Range("A3:H" & w行 + 2) = 二次元配列
    End If

End Sub
