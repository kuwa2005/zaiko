'****************************************************************
'*  <登録>ボタンがクリックされた場合
'****************************************************************
Private Sub 登録_Click()
    
    Call 画面停止有効
    'Call シート保護解除
    '*----------*
    Call DB_OPEN
    Call テーブルを開く("T_在庫マスタ", "コード")
    Call 在庫マスタ登録データチェック
    Call テーブルを閉じる
    Call DB_CLOSE
    '*----------*
    If (エラー = "Y") Then
    Else
        '*----------*
        Call DB_OPEN
        Call テーブルを開く("T_在庫マスタ", "コード")
        Call 在庫マスタ登録データセット
        Call テーブルを閉じる
        Call DB_CLOSE
        '*----------*
        Call データエリア削除("A3", "AZ500000", "A3")
        '*----------*
        Call DB_OPEN
        Call テーブルを開く("T_在庫マスタ", "コード")
        If (エラー = "Y") Then
            Call メッセージ(3)
        Else
            Call 在庫マスタデータ取り込み
            Call 在庫マスタ登録画面クリア
        End If
        Call テーブルを閉じる
        Call DB_CLOSE
        '*----------*
    End If
    'Call シート保護設定
    Call 画面停止無効
    
End Sub

'*  <閉じる>ボタンがクリックされた場合

Private Sub 閉じる_Click()

'================================================================
'=  在庫マスタ登録データセット
'================================================================
Sub 在庫マスタ登録データセット()
    
    With r
        .AddNew
        ![コード] = 未入力の場合Null変換(コード)
        ![品名] = 未入力の場合Null変換(品名)
        ![数量] = 数量
        ![単位] = 未入力の場合Null変換(単位)
        ![単価] = 未入力の場合Null変換(単価)
        ![残数量] = 残数量
        ![安全在庫数] = スペースの場合ゼロ変換(安全在庫数)
        ![最小発注数量] = スペースの場合ゼロ変換(最小発注数量)
        ![棚番] = 未入力の場合Null変換(棚番)
        ![標準納入日数] = 未入力の場合Null変換(標準納入日数)
        ![取引先] = 未入力の場合Null変換(取引先)
        ![備考] = 未入力の場合Null変換(備考)
        ![登録者] = Sheets("メイン").Range("N5")
        ![登録日] = Format(Now, "yyyy/mm/dd")
        .Update
    End With

End Sub

'================================================================
'=　在庫マスタ登録データチェック
'================================================================
Sub 在庫マスタ登録データチェック()
    
    エラー = "N"
    
    '*----------*
    コード = 半角変換(大文字変換(Trim(コード)))
    If (コード = "") Then
        MsgBox "コードを入力して下さい。", 16, "メッセージ"
        コード.SetFocus
        エラー = "Y"
        Exit Sub
    End If
    '*----------*
    r.FindFirst "コード = '" & コード & "'"
    If (Not r.NoMatch) Then
        MsgBox "対象のコードは既に在庫マスタに登録されています。", 16, "メッセージ"
        コード.SetFocus
        エラー = "Y"
        Exit Sub
    End If
    '*----------*
    If (品名 = "") Then
        MsgBox "品名を入力して下さい。", 16, "メッセージ"
        品名.SetFocus
        エラー = "Y"
        Exit Sub
    End If
    '*----------*
    If (Trim(数量) = "" Or 数量 = 0) Then
        MsgBox "基本数量を入力して下さい。", 16, "メッセージ"
        数量.SetFocus
        エラー = "Y"
        Exit Sub
    End If
    '*----------*
    If (Trim(単位) = "") Then
        MsgBox "単位を入力して下さい。", 16, "メッセージ"
        単位.SetFocus
        エラー = "Y"
        Exit Sub
    End If
    '*----------*
    If (Trim(単価) = "" Or 単価 = 0) Then
        MsgBox "単価を入力して下さい。", 16, "メッセージ"
        単価.SetFocus
        エラー = "Y"
        Exit Sub
    End If
    '*----------*
    If (Trim(残数量) = "") Then
        MsgBox "残数量を入力して下さい。", 16, "メッセージ"
        残数量.SetFocus
        エラー = "Y"
        Exit Sub
    End If
    '*----------*
    棚番 = 半角変換(大文字変換(Trim(棚番)))
    '*----------*
    
End Sub

'================================================================
'=　在庫マスタ登録画面を閉じる

Sub 在庫マスタ登録画面を閉じる()
    
    Range("A3").Select
    Unload 在庫マスタ登録画面

End Sub
