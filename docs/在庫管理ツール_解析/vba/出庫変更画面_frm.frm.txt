'****************************************************************
'*  画面がアクティブになった場合
'****************************************************************
Private Sub UserForm_Activate()
    
    '*----------*
    管理NO = ""
    コード = ""
    品名 = ""
    出庫数量 = ""
    備考 = ""
    '*----------*
    管理NO = w管理NO
    コード = wコード
    品名 = w品名
    出庫数量 = w数量
    備考 = w備考
    '*----------*
    出庫数量.SetFocus

End Sub

'****************************************************************
'*  <削除>ボタンがクリックされた場合
'****************************************************************
Private Sub 削除_Click()
    
    Call 画面停止有効
    'Call シート保護解除
    If MsgBox("削除してもよろしいですか？", 292, "メッセージ") = 6 Then
        '*----------*
        Call DB_OPEN
        Call データ削除("T_出庫データ", "管理NO", 管理NO)
        Call DB_CLOSE
        '*----------*
        Call データエリア削除("A3", "AZ500000", "A3")
        '*----------*
        Call DB_OPEN
        Call テーブルを開く("T_出庫データ", "管理NO desc")
        If (エラー = "Y") Then
        Else
            Call 出庫実績確認データ取り込み
        End If
        Call テーブルを閉じる
        Call DB_CLOSE
        '*----------*
        Call 出庫変更画面を閉じる
    End If
    'Call シート保護設定
    Call 画面停止無効
    
End Sub

'****************************************************************
'*  <変更>ボタンがクリックされた場合
'****************************************************************
Private Sub 変更_Click()
    
    Call 画面停止有効
    'Call シート保護解除
    Call 出庫確認データチェック
    If (エラー = "Y") Then
    Else
        '*----------*
        Call DB_OPEN
        Call テーブルを開く("T_出庫データ", "管理NO desc")
        Call 出庫確認データセット
        Call テーブルを閉じる
        Call DB_CLOSE
        '*----------*
        Call データエリア削除("A3", "AZ500000", "A3")
        '*----------*
        Call DB_OPEN
        Call テーブルを開く("T_出庫データ", "管理NO desc")
        If (エラー = "Y") Then
        Else
            Call 出庫実績確認データ取り込み
        End If
        Call テーブルを閉じる
        Call DB_CLOSE
        '*----------*
        Call 出庫変更画面を閉じる
    End If
    'Call シート保護設定
    Call 画面停止無効
    
End Sub

'****************************************************************
'*  <閉じる>ボタンがクリックされた場合
'****************************************************************
Private Sub 閉じる_Click()
    
    Call 出庫変更画面を閉じる
    
End Sub

'================================================================
'=　出庫変更画面を閉じる
'================================================================
Sub 出庫変更画面を閉じる()
    
    Range("A3").Select
    Unload 出庫変更画面
    
End Sub

'================================================================
'=　出庫確認データチェック
'================================================================
Sub 出庫確認データチェック()
    
    エラー = "N"
    
    If (IsNull(出庫数量) = True Or Trim(出庫数量) = "" Or 出庫数量 = 0) Then
        MsgBox "出庫数量を入力して下さい。", 16, "メッセージ"
        出庫数量.SetFocus
        エラー = "Y"
        Exit Sub
    End If

End Sub

'================================================================
'=  出庫確認データセット
'================================================================
Sub 出庫確認データセット()
    
    With r
        .FindFirst "管理NO = '" & 管理NO & "'"
        If (Not .NoMatch) Then
            .Edit
            ![コード] = 未入力の場合Null変換(コード)
            ![コード] = 未入力の場合Null変換(コード)
            ![品名] = 未入力の場合Null変換(品名)
            ![出庫数] = 未入力の場合Null変換(出庫数量)
            ![備考] = 未入力の場合Null変換(備考)
            .Update
        End If
    End With

End Sub
