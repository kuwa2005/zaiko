'****************************************************************
'*　<保存>ボタン
'****************************************************************
Sub 保存ボタン()
    
    Call 画面停止有効
    'Call シート保護解除
    Call オートフィルタ解除("入庫")
    Call オートフィルタ設定("A2", "O2", "A3")
    Call 最終行チェック(2, 2, 2)
    If (存在チェック = "N") Then
        Call メッセージ(2)
    Else
        Call 保存対象データチェック
        If (エラー = "Y") Then
        Else
            '*----------*
            Call DB_OPEN
            Call テーブルを開く("T_在庫発注データ", "管理NO")
            Call 入庫者及び入庫日セット
            Call テーブルを閉じる
            Call DB_CLOSE
            '*----------*
            Call データエリア削除("A3", "AZ500000", "A3")
            '*----------*
            Call DB_OPEN
            Call テーブルを開く("T_在庫発注データ", "管理NO desc")
            If (エラー = "Y") Then
            Else
                Call 入庫データ取り込み
                Call 最終行チェック(1, 1, 2)
                Call プルダウンセット("B3", "B" & 最終行, "入庫,入庫取消,削除", "A3")
            End If
            Call テーブルを閉じる
            Call DB_CLOSE
            '*----------*
        End If
    End If
    'Call シート保護設定
    Call 画面停止無効
    
End Sub

'****************************************************************
'*　<削除>ボタン
'****************************************************************
Sub 削除ボタン_入庫()
    
    Call 画面停止有効
    'Call シート保護解除
    Call オートフィルタ解除("入庫")
    Call オートフィルタ設定("A2", "O2", "A3")
    Call 最終行チェック(1, 16, 2)
    If (存在チェック = "N") Then
        Call メッセージ(1)
    Else
        Call 削除対象データチェック
        If (エラー = "Y") Then
        Else
            If MsgBox("選択データを削除しますか？", 292, "メッセージ") = 6 Then
                Call 画面停止有効
                '*----------*
                Call DB_OPEN
                Call 入庫データ削除
                Call DB_CLOSE
                '*----------*
                Call データエリア削除("A3", "AZ500000", "A3")
                '*----------*
                Call DB_OPEN
                Call テーブルを開く("T_在庫発注データ", "管理NO desc")
                If (エラー = "Y") Then
                Else
                    Call 入庫データ取り込み
                    Call 最終行チェック(1, 1, 2)
                    Call プルダウンセット("B3", "B" & 最終行, "入庫,入庫取消,削除", "A3")
                End If
                Call テーブルを閉じる
                Call DB_CLOSE
                '*----------*
                Call 画面停止無効
            End If
        End If
    End If
    'Call シート保護設定
    Call 画面停止無効
    
End Sub

'================================================================
'=  保存対象データチェック
'================================================================
Sub 保存対象データチェック()
    
    エラー = "N"
    
    For w行 = 3 To 最終行
        If (Trim(Range("B" & w行)) = "削除") Then
            MsgBox "内容欄に「削除」が入力されています。", 16, "メッセージ"
            エラー = "Y"
            Exit Sub
        End If
    Next

End Sub

'================================================================
'=  削除対象データチェック
'================================================================
Sub 削除対象データチェック()
    
    エラー = "N"
    
    For w行 = 3 To 最終行
        Select Case Trim(Range("B" & w行))
            Case "入庫"
                MsgBox "内容欄に「入庫」が入力されています。", 16, "メッセージ"
                エラー = "Y"
                Exit Sub
            Case "入庫取消"
                MsgBox "内容欄に「入庫取消」が入力されています。", 16, "メッセージ"
                エラー = "Y"
                Exit Sub
        End Select
    Next

End Sub

'================================================================
'=　入庫データ削除
'================================================================
Sub 入庫データ削除()
    
    For w行 = 3 To 最終行
        If (Trim(Range("B" & w行)) = "削除") Then
            sql = "delete * from T_在庫発注データ where 管理NO  = '" & Trim(Range("N" & w行)) & "'"
            cn.Execute (sql)
        End If
    Next

End Sub

'================================================================
'=  入庫者及び入庫日セット
'================================================================
Sub 入庫者及び入庫日セット()
    
    With r
        For w行 = 3 To 最終行
            w管理NO = Range("N" & w行)
            Select Case Trim(Range("B" & w行))
                '*----------*
                Case "入庫"
                    If (Trim(Range("C" & w行)) = "") Then
                        .FindFirst "管理NO = '" & w管理NO & "'"
                        If (Not .NoMatch) Then
                            .Edit
                            ![入庫者] = Sheets("メイン").Range("N5")
                            ![入庫日] = Format(Now, "yyyy/mm/dd")
                            .Update
                        End If
                    End If
                '*----------*
                Case "入庫取消"
                    If (Trim(Range("C" & w行)) <> "") Then
                        .FindFirst "管理NO = '" & w管理NO & "'"
                        If (Not .NoMatch) Then
                            .Edit
                            ![入庫者] = Null
                            ![入庫日] = Null
                            .Update
                        End If
                    End If
            End Select
        Next
    End With

End Sub

'================================================================
'=  入庫データ取り込み
'================================================================
Sub 入庫データ取り込み()

    Dim 二次元配列 As Variant
    
    エラー = "N"
    
    If (r.EOF) Then
        エラー = "Y"
        Exit Sub
    End If
    
    ReDim 二次元配列(全件数 - 1, 14)
        
    w行 = 0
    With r
        .MoveFirst
        Do Until (.EOF)
            二次元配列(w行, 0) = w行 + 1
            If (IsNull(![入庫者]) = True) Then
            Else
                二次元配列(w行, 2) = "入庫済み"
            End If
            二次元配列(w行, 3) = ![入庫日]
            二次元配列(w行, 4) = ![コード]
            二次元配列(w行, 5) = ![品名]
            二次元配列(w行, 6) = ![数量]
            二次元配列(w行, 7) = ![納期]
            二次元配列(w行, 8) = ![型式]
            二次元配列(w行, 9) = ![依頼者]
            二次元配列(w行, 10) = ![依頼日]
            二次元配列(w行, 11) = ![受付者]
            二次元配列(w行, 12) = ![受付日]
            二次元配列(w行, 13) = ![管理NO]
            二次元配列(w行, 14) = ![備考]
            w行 = w行 + 1
            .MoveNext
        Loop
    End With
    If (w行 <= 1) Then
        Range("A3:O3") = 二次元配列
    Else
        Range("A3:O" & w行 + 2) = 二次元配列
    End If

End Sub
