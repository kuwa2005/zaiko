'****************************************************************
'*  画面がアクティブになった場合
'****************************************************************
Private Sub UserForm_Activate()
    
    Dim w日 As Long
    Dim w最終行 As Long
    Dim w範囲 As String
    Dim 開始日 As Date
    Dim 終了日 As Date
        
    '*----------*
    品名 = ""
    コード = ""
    数量 = ""
    納期.Clear
    型式 = ""
    備考 = ""
    最小発注数量 = ""
    '*----------*
    '（コード）
    With コード
      .ColumnHeads = True
      .ColumnCount = 2
      .ColumnWidths = "50;200"
      .ListWidth = "250"
    End With
    w最終行 = Sheets("在庫マスタ").Range("A" & Rows.Count).End(xlUp).Row
    If (w最終行 = 2) Then
        w範囲 = "在庫マスタ!B3:C3"
    Else
        w範囲 = "在庫マスタ!B3:C" & w最終行
    End If
    コード.RowSource = w範囲
    '（納期）
    開始日 = Now() - 300
    終了日 = Now() + 300
    With 納期
        For w日 = 開始日 To 終了日
            .AddItem Format(w日, "yyyy/mm/dd")
        Next
    End With
    納期 = Format(Now, "yyyy/mm/dd")
    '*----------*
    コード.SetFocus

End Sub

'****************************************************************
'*  <登録>ボタン
'****************************************************************
Private Sub 登録_Click()
    
    Call 画面停止有効
    Call シート保護解除
    '*----------*
    Call DB_OPEN
    Call テーブルを開く("T_在庫マスタ", "コード")
    Call 発注登録データチェック
    Call テーブルを閉じる
    Call DB_CLOSE
    '*----------*
    If (エラー = "Y") Then
    Else
        '*----------*
        Call DB_OPEN
        Call テーブルを開く("T_採番テーブル", "手配管理NO")
        Call 手配管理NO採番
        Call テーブルを閉じる
        Call DB_CLOSE
        '*----------*
        Call 最終行チェック(3, 13, 2)
        Call 発注登録データセット
        Call 発注登録画面クリア
    End If
    Call シート保護設定
    Call 画面停止無効
    
End Sub

'*  <クリア>ボタン
'****************************************************************

    Call 発注登録画面クリア

End Sub

'****************************************************************
'*  <閉じる>ボタン

Private Sub 閉じる_Click()

'================================================================
'=  発注登録画面閉じる
'================================================================
Sub 発注登録画面閉じる()
    
    Range("A3").Select
    Unload 発注登録画面

End Sub

'================================================================
'=  納期欄が変更された場合
'================================================================
Private Sub 納期_Change()
    
    If (納期 = "") Then
    Else
        型式.SetFocus
    End If

End Sub

'================================================================
'=  コード欄が変更された場合
'================================================================
Private Sub コード_Change()
    
    If (コード = "") Then
    Else
        Call DB_OPEN
        Call テーブルを開く("T_在庫マスタ", "コード")
        Call 品名及び最小発注数量セット
        Call テーブルを閉じる
        Call DB_CLOSE
        数量.SetFocus
    End If

End Sub

'================================================================
'=  品名及び最小発注数量セット
'================================================================
Sub 品名及び最小発注数量セット()

    With r
        .FindFirst "コード = '" & コード & "'"
        If (Not .NoMatch) Then
            品名 = ![品名]
            最小発注数量 = "最小発注数量：" & ![最小発注数量]
        End If
    End With
        
End Sub

'================================================================
'=  発注登録データチェック
'================================================================
Sub 発注登録データチェック()
    
    エラー = "N"
    
    '*----------*
    If (コード = "") Then
        MsgBox "コードが選択されていません。", 16, "メッセージ"
        コード.SetFocus
        エラー = "Y"
        Exit Sub
    End If
    '*----------*
    If (数量 = "" Or IsNumeric(数量) = False) Then
        MsgBox "発注数量が入力されていません。", 16, "メッセージ"
        数量.SetFocus
        エラー = "Y"
        Exit Sub
    End If
    '*----------*
    With r
        .FindFirst "コード = '" & コード & "'"
        If (Not .NoMatch) Then
            If (IsNull(![最小発注数量]) = True Or ![最小発注数量] = 0) Then
            Else
                w数量 = Str(数量)
                If (w数量 < ![最小発注数量]) Then
                    MsgBox "発注数量 < 最小発注数量数です。" & Chr(13) & "発注数量を最小発注数量数以上で発注依頼して下さい。", 16, "メッセージ"
                    数量.SetFocus
                    エラー = "Y"
                    Exit Sub
                End If
            End If
        End If
    End With
    '*----------*
    If (納期 = "") Then
        MsgBox "納期が入力されていません。", 16, "メッセージ"
        納期.SetFocus
        エラー = "Y"
        Exit Sub
    End If
    '*----------*
    
End Sub

'================================================================
'=  発注登録画面クリア
'================================================================
Sub 発注登録画面クリア()
    
    品名 = ""
    コード = ""
    数量 = ""
    納期 = ""
    型式 = ""
    備考 = ""
    納期 = Format(Now, "yyyy/mm/dd")
    最小発注数量 = ""
    コード.SetFocus

End Sub

'================================================================
'*  手配管理NO採番
'================================================================
Sub 手配管理NO採番()
    
    With r
        .MoveFirst
        w管理NO = ![手配管理NO] + 1
        .Edit
        ![手配管理NO] = w管理NO
        .Update
    End With

End Sub
        
'================================================================
'*  発注登録データセット
'================================================================
Sub 発注登録データセット()
    
    Range("A" & 最終行 + 1) = 最終行 - 1
    Range("C" & 最終行 + 1) = コード
    Range("D" & 最終行 + 1) = 品名
    Range("E" & 最終行 + 1) = Str(数量)
    Range("F" & 最終行 + 1) = Format(納期, "yyyy/mm/dd")
    Range("G" & 最終行 + 1) = 型式
    Range("H" & 最終行 + 1) = Sheets("メイン").Range("N5")
    Range("I" & 最終行 + 1) = Format(Now, "yyyy/mm/dd")
    Range("J" & 最終行 + 1) = CStr("H" & Format(w管理NO, "0000000000"))
    Range("K" & 最終行 + 1) = 備考

End Sub
